```ts
// Every note reachable at /note/<slug> — listed AND unlisted (SEO sitemap
// scope; the listed flag gates only the browsable library, not access).
export async function getAllNotesForSitemap(): Promise<
  Array<{ slug: string; lastModified: string | null }>
> {
  if (!db) {
    console.warn("Firebase not initialized");
    return [];
  }

  try {
    const notesRef = collection(db, "notes");
    const querySnapshot = await getDocs(query(notesRef));
    return querySnapshot.docs
      .map((d) => {
        const data = d.data() as StudyGuide;
        return {
          slug: data.slug || d.id,
          lastModified: data.updatedAt ?? data.createdAt ?? null,
        };
      })
      .filter((n) => n.slug);
  } catch (error) {
    console.error("Error fetching notes for sitemap:", error);
    return [];
  }
}
```

- [ ] **Step 2: Create `app/sitemap.ts`**

```ts
import type { MetadataRoute } from "next";
import { getAllNotesForSitemap } from "@/lib/studyGuide";

const SITE_URL = "https://quizliai.com";

// ISR: regenerated at most hourly on traffic, so newly published notes
// appear without a redeploy.
export const revalidate = 3600;

export default async function sitemap(): Promise<MetadataRoute.Sitemap> {
  const staticEntries: MetadataRoute.Sitemap = [
    {
      url: SITE_URL,
      changeFrequency: "weekly",
      priority: 1,
    },
  ];

  const notes = await getAllNotesForSitemap();
  const noteEntries: MetadataRoute.Sitemap = notes.map((n) => ({
    url: `${SITE_URL}/note/${n.slug}`,
    lastModified: n.lastModified ? new Date(n.lastModified) : undefined,
    changeFrequency: "weekly",
    priority: 0.7,
  }));

  return [...staticEntries, ...noteEntries];
}
```

- [ ] **Step 3: Create the new `app/note/[slug]/page.tsx`**

```tsx
import type { Metadata } from "next";
import { getStudyGuide } from "@/lib/studyGuide";
import { GuideClient } from "./GuideClient";

const SITE_URL = "https://quizliai.com";

type PageProps = { params: Promise<{ slug: string }> };

export async function generateMetadata({ params }: PageProps): Promise<Metadata> {
  const { slug } = await params;
  const guide = await getStudyGuide(slug);

  if (!guide) {
    return {
      title: "Study Guide | QuizliAI",
      description: "AI-generated study guide with notes, flashcards, and quiz.",
    };
  }

  const description =
    guide.tldr && guide.tldr.length > 0
      ? guide.tldr[0].slice(0, 160)
      : "AI-generated study guide with notes, flashcards, and quiz.";

  return {
    title: `${guide.title} | QuizliAI`,
    description,
    alternates: { canonical: `${SITE_URL}/note/${slug}` },
    openGraph: {
      title: guide.title,
      description,
      ...(guide.youtubeId
        ? { images: [{ url: `https://img.youtube.com/vi/${guide.youtubeId}/maxresdefault.jpg` }] }
        : {}),
    },
  };
}

export default async function PublicStudyGuidePage({ params }: PageProps) {
  const { slug } = await params;
  const guide = await getStudyGuide(slug);
  return <GuideClient slug={slug} initialGuide={guide} />;
}
```

(Note: `params` is a **Promise** in this Next version — `await params` is the current convention; if tsc flags the type, consult `node_modules/next/dist/docs/01-app/03-api-reference/04-functions/generate-metadata.md` for the exact signature and adjust the `PageProps` type to match, keeping the `await`.)

import { writeFile } from "fs/promises";

const URL = "https://apps.apple.com/us/app/quizliai/id6751740981";

const svg = await QRCode.toString(URL, {
  type: "svg",
  margin: 1,
  width: 240,
  color: { dark: "#000000", light: "#FFFFFF" },
